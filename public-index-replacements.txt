5) REPLACE THE ENTIRE queueUnresolvedBooksForReleaseRecheck FUNCTION WITH:

function queueUnresolvedBooksForReleaseRecheck(){
  try{
    if(localStorage.getItem(UNRESOLVED_RECHECK_KEY)===UNRESOLVED_RECHECK_VERSION)return 0;
    let queued=0;
    for(const book of Object.values(data.books||{})){
      // Never touch a book that already has confirmed AR data.
      if(book.lookupStatus!=='no_ar' && book.lookupStatus!=='error')continue;
      book.recheckFromStatus=book.lookupStatus;
      book.recheckFromVersion=book.lastLookupVersion||null;
      book.recheckRelease=UNRESOLVED_RECHECK_VERSION;
      book.lookupStatus='queued';
      book.error=null;
      book.forceRefresh=true;
      queued++;
    }
    // Mark this release as processed even if there was nothing to retry.
    localStorage.setItem(UNRESOLVED_RECHECK_KEY,UNRESOLVED_RECHECK_VERSION);
    if(queued){
      localStorage.setItem(KEY,JSON.stringify(data));
      trackEvent('unresolved_books_recheck_queued',{
        count:queued,
        release:UNRESOLVED_RECHECK_VERSION,
        toVersion:APP_VERSION
      });
    }
    return queued;
  }catch{
    return 0;
  }
}

async function lookupBook(book,refresh=false){
  const recheckRelease=book.recheckRelease||null;
  const statusBeforeLookup=book.recheckFromStatus||book.retryFromStatus||book.lookupStatus;
  const versionBeforeLookup=book.recheckFromVersion||book.lastLookupVersion||null;
  const lookupTrigger=recheckRelease?'version_recheck':(book.lookupTrigger||(refresh?'manual_retry':'normal'));
  activeLookups++;
  const telemetryLookupStarted=performance.now();
  trackEvent('lookup_started',{
    isbn:book.isbn,
    refresh:Boolean(refresh||book.forceRefresh),
    trigger:lookupTrigger,
    priorStatus:statusBeforeLookup,
    priorVersion:versionBeforeLookup,
    recheckRelease
  });
  book.lookupStatus='pending';

11) REPLACE THE ONE-LINE retry FUNCTION WITH:

function retry(isbn){
  const b=data.books[isbn];
  if(!b)return;
  const priorStatus=b.lookupStatus;
  trackEvent('lookup_retry',{isbn,priorStatus});
  b.retryFromStatus=priorStatus;
  b.lookupTrigger='manual_retry';
  b.lookupStatus='queued';
  b.error=null;
  persist();
  renderAll();
  pumpQueue();
}
